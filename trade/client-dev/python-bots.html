NadaTrade Documentation

Building Bots: Python

Connect to NadaTrade from Python using either FIX 4.2 (raw TCP) or WebSocket (JSON). Both gateways produce identical order flow — the choice is about wire format and library availability, not capability.

Choosing a Gateway

FIX 4.2WebSocket
TransportRaw TCP, port 9823WebSocket, port 9824
AuthLogon (35=A) with username/passwordJWT HS256 token as query param
FramingSOH-delimited tag=value, BodyLength, CheckSumJSON text frames
LibraryStandard library only (socket)websockets (pip install websockets)
Best forExisting FIX infrastructure, lowest latencyQuick prototyping, browser clients
Market data35=V (subscribe), 35=W/X (snapshots/updates)subscribe/unsubscribe messages

Prerequisites

  • Python 3.9+
  • NadaTrade running on your server (or localhost)
  • FIX track: no dependencies beyond standard library
  • WS track: pip install websockets
  • JWT generation (WS): pip install pyjwt (only needed to generate tokens; not for the bot itself)

Track A: FIX 4.2 Bot

Connection Lifecycle

Every FIX session follows this sequence:

TCP connect → Logon (35=A) → Orders (35=D/F/G) → Logout (35=5) → TCP close

Until Logon succeeds, any non-A message is rejected with a SessionReject (35=3).

FIX Message Framing

Every FIX message is structured as:

8=FIX.4.2\x019=<body-length>\x01<tag=value pairs ending with \x01>10=<checksum>\x01
  • SOH (\x01) is the field delimiter
  • BodyLength (tag 9) counts bytes from tag 35 to the delimiter before tag 10
  • CheckSum (tag 10) sums all preceding bytes modulo 256, zero-padded to 3 digits

Helper Functions

import socket, datetime

SOH = "\x01"

def checksum(msg):
    """Sum all bytes modulo 256."""
    return sum(msg.encode("ascii")) % 256

def fix_msg(body):
    """Wrap body with header, BodyLength, and CheckSum."""
    msg = f"8=FIX.4.2{SOH}9={len(body)}{SOH}{body}"
    return f"{msg}10={checksum(msg):03d}{SOH}"

def recv_fix(sock):
    """Read one FIX message delimited by trailing SOH."""
    buf = b""
    while not buf.endswith(b"\x01"):
        chunk = sock.recv(4096)
        if not chunk:
            return None
        buf += chunk
    return buf.decode("ascii")

def parse_tags(msg):
    """Parse FIX tag=value pairs into a dict."""
    return dict(pair.split("=", 1) for pair in msg.split(SOH) if "=" in pair)

Logon

sock = socket.socket(socket.AF_INET, socket.SOCK_STREAM)
sock.connect(("your-server.example.com", 9823))

now = datetime.datetime.utcnow().strftime("%Y%m%d-%H:%M:%S")
logon_body = (
    f"35=A{SOH}"
    f"49=MY-BOT{SOH}"
    f"56=NADATRADE{SOH}"
    f"34=1{SOH}"
    f"52={now}{SOH}"
    f"98=0{SOH}"
    f"108=30{SOH}"
    f"553=alice{SOH}"       # your FIX username
    f"554=alice-pass{SOH}"  # your FIX password
)
sock.sendall(fix_msg(logon_body).encode("ascii"))

resp = parse_tags(recv_fix(sock))
assert resp.get("35") == "A", f"Logon failed: {resp.get('58', 'unknown error')}"
print(f"Connected: heartbeat={resp.get('108')}s")

Reading Execution Reports

def read_responses(sock, count=3):
    """Read and display ExecutionReports."""
    for _ in range(count):
        msg = recv_fix(sock)
        tags = parse_tags(msg)
        msg_type = tags.get("35")
        if msg_type == "8":  # ExecutionReport
            print(f"  ER: exec_type={tags.get('150')} "
                  f"ord_status={tags.get('39')} "
                  f"cum={tags.get('14')} leaves={tags.get('151')} "
                  f"text={tags.get('58', '')}")
        elif msg_type == "9":  # CancelReject
            print(f"  CancelRejected: {tags.get('58')}")
        elif msg_type == "0":  # Heartbeat
            pass  # ignore heartbeats
        else:
            print(f"  MsgType={msg_type}: {tags}")

read_responses(sock, 5)

Track B: WebSocket Bot

JWT Token

The WebSocket gateway authenticates with HS256 JWT tokens. You need a token with the sub claim set to your participant comp_id:

import jwt, time

secret = "your-server-jwt-secret"  # same as ws_jwt_secret in server config
token = jwt.encode(
    {"sub": "ALICE", "exp": int(time.time()) + 31536000},
    secret, algorithm="HS256"
)

Pass the token as a query parameter: ws://host:9824/v1/orders?token=<token>.

Install Dependencies

pip install websockets

Connect and Send an Order

import asyncio, json
import websockets

TOKEN = "<your-jwt-token>"
HOST = "your-server.example.com"

async def main():
    uri = f"ws://{HOST}:9824/v1/orders?token={TOKEN}"

    async with websockets.connect(uri) as ws:
        # Server sends a welcome message first
        welcome = json.loads(await ws.recv())
        print(f"Connected: {welcome['message']}")

        # Send a limit order
        order = {
            "type": "new_order",
            "req_id": "r1",
            "instrument": "PETR4",
            "side": "buy",
            "order_type": "limit",
            "qty": "100",
            "price": "50.00",
            "tif": "day",
            "client_order_id": "my-order-1",
        }
        await ws.send(json.dumps(order))

        # Read responses
        while True:
            msg = json.loads(await ws.recv())
            t = msg["type"]
            print(f"  {t} (seq={msg.get('seq')})")

            if t == "fill" and msg.get("is_final"):
                break
            if t == "order_rejected":
                print(f"  Rejected: {msg['reason']}")
                break

asyncio.run(main())

Handling the Response Stream

Every server message (except error and unsubscribed) carries seq and sim_time. The seq field is a global monotonic counter — use it to order events across instruments.

async def handle_responses(ws):
    while True:
        msg = json.loads(await ws.recv())
        t = msg["type"]

        handlers = {
            "order_accepted": lambda m: print(f"Accepted: {m['order_id']}",
            "order_rejected": lambda m: print(f"Rejected: {m['reason']}"),
            "fill": lambda m: print(f"Fill: {m['fill_qty']} @ {m['fill_price']} {'FINAL' if m['is_final'] else 'partial'}"),
            "order_cancelled": lambda m: print(f"Cancelled: {m['reason']}"),
            "order_expired": lambda m: print(f"Expired"),
            "order_replaced": lambda m: print(f"Replaced: {m['order_id']} -> {m['new_order_id']}"),
            "cancel_rejected": lambda m: print(f"Cancel rejected: {m['reason']}"),
            "error": lambda m: print(f"Error: {m['code']} — {m['message']}"),
        }

        handler = handlers.get(t)
        if handler:
            handler(msg)

    async with websockets.connect(uri) as ws:
        welcome = json.loads(await ws.recv())

        }))
        await ws.send(json.dumps({
            "type": "subscribe", "req_id": "r2",
            "instrument": "PETR4", "channel": "orderbook"
        }))

        while True:
            msg = json.loads(await ws.recv())
            t = msg["type"]

            if t == "trade":
                print(f"Trade: {msg['instrument']} @ {msg['price']} qty={msg['qty']} side={msg['side']}")
            elif t == "orderbook_snapshot":
                print(f"Book: bids={msg['bids'][:3]} asks={msg['asks'][:3]}")
            elif t == "orderbook_update":
                print(f"Book update: bids={msg['bids']} asks={msg['asks']} deletes={msg['deletes']}")
            elif t == "session":
                print(f"Session: {msg['instrument']} -> {msg['state']}")

Error Handling Patterns

async def safe_send(ws, msg):
    """Send with error handling for server-side errors."""
    await ws.send(json.dumps(msg))

    # The immediate ack for new/cancel/replace comes as an error-type
    # with codes like "cancel_sent" or "replace_sent"
    resp = json.loads(await ws.recv())
    if resp["type"] == "error":
        if resp["code"] in ("cancel_sent", "replace_sent"):
            return  # These are expected acks, not errors
        raise RuntimeError(f"Server error: {resp['code']} — {resp['message']}")
    return resp

Complete Bot Template

Save as bot.py:

#!/usr/bin/env python3
"""NadaTrade WebSocket bot template."""
import asyncio, json, os
import websockets

HOST = os.environ.get("NADATRADE_HOST", "localhost")
TOKEN = os.environ.get("NADATRADE_TOKEN", "your-token")

async def run():
    uri = f"ws://{HOST}:9824/v1/orders?token={TOKEN}"

    async with websockets.connect(uri) as ws:
        welcome = json.loads(await ws.recv())
        print(welcome["message"])

        # --- Your strategy here ---

        await ws.send(json.dumps({
            "type": "new_order",
            "req_id": "r1",
            "instrument": os.environ.get("INSTRUMENT", "PETR4"),
            "side": "buy",
            "order_type": "limit",
            "qty": "100",
            "price": "50.00",
            "tif": "day",
        }))

        async for raw in ws:
            msg = json.loads(raw)
            t = msg["type"]

            if t == "order_accepted":
                print(f"Accepted: {msg['order_id'][:8]}...")
            elif t == "fill":
                print(f"Fill: {msg['fill_qty']} @ {msg['fill_price']}")
                if msg.get("is_final"):
                    break
            elif t == "order_rejected":
                print(f"Rejected: {msg['reason']}")
                break

    print("Done.")

if __name__ == "__main__":
    asyncio.run(run())
